>>     private static void applyForJob() {
>>         String studentId = readText("Student ID: ");
>>         ObjectId jobId = parseObjectId(readText("Job MongoDB ID: "));
>>
>>         if (jobId == null) {
>>             System.out.println("Invalid job ID.");
>>             return;
>>         }
>>
>>         Document student = students.find(eq("studentId", studentId)).first();
>>         Document job = jobs.find(eq("_id", jobId)).first();
>>
>>         if (student == null) {
>>             System.out.println("Student not found. Register first.");
>>             return;
>>         }
>>
>>         if (job == null || !"OPEN".equals(job.getString("status"))) {
>>             System.out.println("Job not found or is closed.");
>>             return;
>>         }
>>
>>         if (student.getBoolean("placed", false)) {
>>             System.out.println("This student is already marked as placed.");
>>             return;
>>         }
>>
>>         double cgpa = student.getDouble("cgpa");
>>         double minCgpa = job.getDouble("minCgpa");
>>         String eligibleDepartment = job.getString("department");
>>         String studentDepartment = student.getString("department");
>>
>>         if (cgpa < minCgpa) {
>>             System.out.println("Not eligible: CGPA is below the requirement.");
>>             return;
>>         }
>>
>>         if (!"ALL".equalsIgnoreCase(eligibleDepartment)
>>                 && !eligibleDepartment.equalsIgnoreCase(studentDepartment)) {
>>             System.out.println("Not eligible for this department.");
>>             return;
>>         }
>>
>>         ObjectId studentObjectId = student.getObjectId("_id");
>>
>>         Document duplicate = applications.find(and(
>>                 eq("studentId", studentId),
>>                 eq("jobDriveId", jobId.toHexString())
>>         )).first();
>>
>>         if (duplicate != null) {
>>             System.out.println("This student has already applied for this job.");
>>             return;
>>         }
>>
>>         Document application = new Document("_id", new ObjectId())
>>                 .append("studentId", studentId)
>>                 .append("studentName", student.getString("name"))
>>                 .append("studentObjectId", studentObjectId.toHexString())
>>                 .append("jobDriveId", jobId.toHexString())
>>                 .append("companyName", job.getString("companyName"))
>>                 .append("role", job.getString("role"))
>>                 .append("status", "APPLIED");
>>
>>         applications.insertOne(application);
>>         System.out.println("Application submitted successfully.");
>>         System.out.println("Application MongoDB ID: "
>>                 + application.getObjectId("_id"));
>>     }
>>
>>     private static void updateApplicationStatus() {

>>         if (id == null) {
